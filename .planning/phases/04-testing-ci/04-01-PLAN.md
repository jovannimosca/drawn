    ```kotlin
                    classes("com.example.drawn.MainActivity")
                    classes("com.example.drawn.DrawnApplication")
                    classes("com.example.drawn.ui.navigation.*")
                    classes("com.example.drawn.ui.theme.*")
                }
            }
            verify {
                rule {
                    minBound(80)
                }
            }
        }
    }
    ```
