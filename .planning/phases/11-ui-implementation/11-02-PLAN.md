```kotlin
package com.example.drawn.ui.deck

import androidx.compose.foundation.clickable
import androidx.compose.foundation.layout.*
import androidx.compose.foundation.lazy.LazyColumn
import androidx.compose.foundation.lazy.items
import androidx.compose.material.icons.Icons
import androidx.compose.material.icons.filled.Add
import androidx.compose.material3.*
import androidx.compose.runtime.*
import androidx.compose.ui.Alignment
import androidx.compose.ui.Modifier
import androidx.compose.ui.unit.dp
import androidx.hilt.navigation.compose.hiltViewModel
import com.example.drawn.domain.model.Deck
import com.example.drawn.ui.theme.*

@Composable
fun DeckListScreen(
    viewModel: DeckViewModel = hiltViewModel(),
    onDeckClick: (Long) -> Unit,
    onAddDeck: () -> Unit
) {
    val uiState by viewModel.customDecks.collectAsStateWithLifecycle()

    Box(modifier = Modifier.fillMaxSize()) {
        when (val state = uiState) {
            is DeckListUiState.Loading -> {
                CircularProgressIndicator(modifier = Modifier.align(Alignment.Center))
            }
            is DeckListUiState.Success -> {
                if (state.decks.isEmpty()) {
                    EmptyDeckState(modifier = Modifier.align(Alignment.Center))
                } else {
                    LazyColumn(
                        modifier = Modifier.fillMaxSize(),
                        contentPadding = PaddingValues(16.dp),
                        verticalArrangement = Arrangement.spacedBy(12.dp)
                    ) {
                        items(state.decks, key = { it.id }) { deck ->
                            DeckItem(
                                deck = deck,
                                onClick = { onDeckClick(deck.id) }
                            )
                        }
                    }
                }
            }
            is DeckListUiState.Error -> {
                ErrorBanner(
                    message = state.message,
                    modifier = Modifier
                )
            }
        }

        // FAB for adding new deck
        FloatingActionButton(
            onClick = onAddDeck,
            modifier = Modifier
                .align(Alignment.BottomEnd)
                .padding(16.dp),
            containerColor = DarkPrimary
        ) {
            Icon(Icons.Default.Add, contentDescription = "Add deck")
        }
    }
}

@Composable
private fun DeckItem(
    deck: Deck,
    onClick: () -> Unit
) {
    Card(
        modifier = Modifier
            .fillMaxWidth()
            .clickable(onClick = onClick),
        colors = CardDefaults.cardColors(containerColor = DarkSurface)
    ) {
        Column(modifier = Modifier.padding(16.dp)) {
            Text(
                text = deck.name,
                style = MaterialTheme.typography.titleMedium,
                color = OnDarkSurface
            )
            if (!deck.description.isNullOrBlank()) {
                Text(
                    text = deck.description,
                    style = MaterialTheme.typography.bodySmall,
                    color = OnDarkSurface.copy(alpha = 0.7f),
                    maxLines = 2
                )
            }
        }
    }
}

@Composable
private fun EmptyDeckState(modifier: Modifier = Modifier) {
    Column(
        modifier = modifier.padding(32.dp),
        horizontalAlignment = Alignment.CenterHorizontally
    ) {
        Text(
            text = "No custom decks yet",
            style = MaterialTheme.typography.titleMedium,
            color = OnDarkSurface
        )
        Spacer(modifier = Modifier.height(8.dp))
        Text(
            text = "Create your first custom deck to get started",
            style = MaterialTheme.typography.bodyMedium,
            color = OnDarkSurface.copy(alpha = 0.7f)
        )
    }
}
```

```kotlin
package com.example.drawn.ui.deck

import androidx.compose.foundation.layout.*
import androidx.compose.foundation.lazy.grid.LazyVerticalGrid
import androidx.compose.foundation.lazy.grid.items
import androidx.compose.material.icons.Icons
import androidx.compose.material.icons.filled.Add
import androidx.compose.material.icons.filled.ArrowBack
import androidx.compose.material.icons.filled.Delete
import androidx.compose.material.icons.filled.Edit
import androidx.compose.material3.*
import androidx.compose.runtime.*
import androidx.compose.ui.Alignment
import androidx.compose.ui.Modifier
import androidx.compose.ui.unit.dp
import androidx.hilt.navigation.compose.hiltViewModel
import com.example.drawn.data.repository.CustomCardRepository
import com.example.drawn.domain.model.CustomCard
import com.example.drawn.ui.theme.*

@Composable
fun DeckDetailScreen(
    deckId: Long,
    onNavigateBack: () -> Unit,
    onEditCard: (CustomCard?) -> Unit
) {
    val cardRepository: CustomCardRepository = hiltViewModel<CardManagementViewModel>().let {
        // Repository injection needed - use hiltViewModel for repo directly
        given cardRepository
    }
    val cards by cardRepository.getCardsForDeck(deckId).collectAsStateWithLifecycle(initial = emptyList())

    Scaffold(
        topBar = {
            TopAppBar(
                title = { Text("Deck Name") }, // TODO: Load from ViewModel
                navigationIcon = {
                    IconButton(onClick = onNavigateBack) {
                        Icon(Icons.Default.ArrowBack, contentDescription = "Back")
                    }
                },
                actions = {
                    IconButton(onClick = { /* Edit deck */ }) {
                        Icon(Icons.Default.Edit, contentDescription = "Edit deck")
                    }
                    IconButton(onClick = { /* Delete deck */ }) {
                        Icon(Icons.Default.Delete, contentDescription = "Delete deck")
                    }
                },
                colors = TopAppBarDefaults.topAppBarColors(
                    containerColor = DarkSurface,
                    titleContentColor = OnDarkSurface,
                    navigationIconContentColor = OnDarkSurface,
                    actionIconContentColor = OnDarkSurface
                )
            )
        },
        floatingActionButton = {
            FloatingActionButton(
                onClick = { onEditCard(null) },
                containerColor = DarkPrimary
            ) {
                Icon(Icons.Default.Add, contentDescription = "Add card")
            }
        }
    ) { padding ->
        if (cards.isEmpty()) {
            Box(
                modifier = Modifier
                    .fillMaxSize()
                    .padding(padding),
                contentAlignment = Alignment.Center
            ) {
                Text(
                    text = "No cards in this deck",
                    color = OnDarkSurface.copy(alpha = 0.7f)
                )
            }
        } else {
            LazyVerticalGrid(
                columns = androidx.compose.foundation.lazy.grid.GridCells.Fixed(2),
                modifier = Modifier
                    .fillMaxSize()
                    .padding(padding),
                contentPadding = PaddingValues(16.dp),
                horizontalArrangement = Arrangement.spacedBy(12.dp),
                verticalArrangement = Arrangement.spacedBy(12.dp)
            ) {
                items(cards, key = { it.id }) { card ->
                    CardThumbnail(
                        card = card,
                        onClick = { onEditCard(card) }
                    )
                }
            }
        }
    }
}

@Composable

                    text = card.keywords,
                    style = MaterialTheme.typography.bodySmall,
                    color = OnDarkSurface.copy(alpha = 0.7f),
                    maxLines = 3
                )
            }
        }
    }
}
```

```kotlin
package com.example.drawn.ui.deck

import androidx.compose.foundation.layout.*
import androidx.compose.material3.*
import androidx.compose.runtime.*
import androidx.compose.ui.Modifier
import androidx.compose.ui.unit.dp
import com.example.drawn.domain.model.CustomCard
import com.example.drawn.ui.theme.*

@Composable
fun CardEditorDialog(
    card: CustomCard? = null, // null = create mode
    onDismiss: () -> Unit,
    onSave: (name: String, imagePath: String?, keywords: String?, meaning: String?, reversedMeaning: String?) -> Unit,
    onDelete: (() -> Unit)? = null
) {
    var name by remember { mutableStateOf(card?.name ?: "") }
    var imagePath by remember { mutableStateOf(card?.imagePath ?: "") }
    var keywords by remember { mutableStateOf(card?.keywords ?: "") }
    var meaning by remember { mutableStateOf(card?.meaning ?: "") }
    var reversedMeaning by remember { mutableStateOf(card?.reversedMeaning ?: "") }

    AlertDialog(
        onDismissRequest = onDismiss,
        title = {
            Text(if (card == null) "Add Card" else "Edit Card")
        },
        text = {
            Column(
                verticalArrangement = Arrangement.spacedBy(12.dp)
            ) {
                OutlinedTextField(
                    value = name,
                    onValueChange = { name = it },
                    label = { Text("Card Name *") },
                    modifier = Modifier.fillMaxWidth(),
                    colors = OutlinedTextFieldDefaults.colors(
                        focusedBorderColor = DarkPrimary,
                        unfocusedBorderColor = OnDarkSurface.copy(alpha = 0.5f)
                    )
                )
                OutlinedTextField(
                    value = keywords,
                    onValueChange = { keywords = it },
                    label = { Text("Keywords") },
                    modifier = Modifier.fillMaxWidth(),
                    minLines = 2,
                    colors = OutlinedTextFieldDefaults.colors(
                        focusedBorderColor = DarkPrimary,
                        unfocusedBorderColor = OnDarkSurface.copy(alpha = 0.5f)
                    )
                )
                OutlinedTextField(
                    value = meaning,
                    onValueChange = { meaning = it },
                    label = { Text("Upright Meaning") },
                    modifier = Modifier.fillMaxWidth(),
                    minLines = 2,
                    colors = OutlinedTextFieldDefaults.colors(
                        focusedBorderColor = DarkPrimary,
                        unfocusedBorderColor = OnDarkSurface.copy(alpha = 0.5f)
                    )
                )
                OutlinedTextField(
                    value = reversedMeaning,
                    onValueChange = { reversedMeaning = it },
                    label = { Text("Reversed Meaning") },
                    modifier = Modifier.fillMaxWidth(),
                    minLines = 2,
                    colors = OutlinedTextFieldDefaults.colors(
                        focusedBorderColor = DarkPrimary,
                        unfocusedBorderColor = OnDarkSurface.copy(alpha = 0.5f)
                    )
                )
            }
        },
        confirmButton = {
            TextButton(
                onClick = {
                    if (name.isNotBlank()) {
                        onSave(name, imagePath.ifBlank { null }, keywords.ifBlank { null }, meaning.ifBlank { null }, reversedMeaning.ifBlank { null })
                    }
                },
                enabled = name.isNotBlank()
            ) {
                Text("Save")
            }
        },
        dismissButton = {
            Row {
                if (card != null && onDelete != null) {
                    TextButton(onClick = onDelete) {
                        Text("Delete", color = DarkError)
                    }
                }
                TextButton(onClick = onDismiss) {
                    Text("Cancel")
                }
            }
        },
        containerColor = DarkSurface,
        titleContentColor = OnDarkSurface,
        textContentColor = OnDarkSurface
    )
}
```
